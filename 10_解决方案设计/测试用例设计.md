```csharp
// TC-HP-U01 示例
[Test]
public void TestApplyDamage_Normal()
{
    var health = new HealthComponent();
    health.MaxHealth = 100f;
    health.MaxShield = 50f;
    health.ShieldAbsorbRate = 0.5f;
    health._Ready();

    health.ApplyDamage(20f);

    Assert.That(health.CurrentHealth, Is.EqualTo(90f));
    Assert.That(health.CurrentShield, Is.EqualTo(40f));
}

// TC-HP-S01 示例
[Test]
public void TestHealthChangedSignalEmitted()
{
    var health = new HealthComponent();
    health.MaxHealth = 100f;
    health.MaxShield = 50f;
    health.ShieldAbsorbRate = 0.5f;
    health._Ready();

    float? receivedHealth = null;
    float? receivedMax = null;
    health.HealthChanged += (cur, max) => { receivedHealth = cur; receivedMax = max; };

    health.ApplyDamage(20f);

    Assert.That(receivedHealth, Is.EqualTo(90f));
    Assert.That(receivedMax, Is.EqualTo(100f));
}
```

```csharp
[Test]
public void TestHurtboxReceivesDamageFromHitbox()
{
    var player = new CharacterBody2D();
    var health = new HealthComponent { MaxHealth = 100f };
    var hurtbox = new HurtboxComponent();
```
